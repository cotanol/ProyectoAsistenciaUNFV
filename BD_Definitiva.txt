-- Eliminar la base de datos 'unfv_bd' si existe y crearla nuevamente
DROP DATABASE IF EXISTS unfv_bd;
CREATE DATABASE unfv_bd;
USE unfv_bd;

-- 1. Crear la tabla 'alumno'
-- Esta tabla almacena la información de los alumnos
CREATE TABLE alumno (
  id_alumno INT AUTO_INCREMENT PRIMARY KEY,  -- Identificador único del alumno (clave primaria)
  codigo_alumno VARCHAR(30) UNIQUE,          -- Código único del alumno asignado por la universidad
  nombres VARCHAR(100),                      -- Nombres del alumno
  apellidos VARCHAR(100)                     -- Apellidos del alumno
);

-- 2. Crear la tabla 'usuario'
-- Esta tabla almacena los usuarios del sistema (administradores y docentes)
CREATE TABLE usuario (
  id_usuario INT AUTO_INCREMENT PRIMARY KEY,  -- Identificador único del usuario (clave primaria)
  nombres VARCHAR(100),                       -- Nombres del usuario
  apellidos VARCHAR(100),                     -- Apellidos del usuario
  tipo_documento ENUM('DNI', 'PASAPORTE'),    -- Tipo de documento de identidad
  nro_documento VARCHAR(20) UNIQUE,           -- Número de documento (único)
  numero VARCHAR(20),                         -- Número de contacto telefónico
  tipo_usuario ENUM('ADMIN', 'DOCENTE'),      -- Tipo de usuario: 'ADMIN' o 'DOCENTE'
  nombre_usuario VARCHAR(100) UNIQUE,         -- Nombre de usuario para el inicio de sesión (único)
  contrasena VARCHAR(100),                    -- Contraseña del usuario
  email VARCHAR(100) UNIQUE                   -- Correo electrónico del usuario (único)
);

-- 3. Crear la tabla 'laboratorio'
-- Esta tabla almacena la información de los laboratorios disponibles
CREATE TABLE laboratorio (
  id_laboratorio INT AUTO_INCREMENT PRIMARY KEY,  -- Identificador único del laboratorio (clave primaria)
  numero_lab VARCHAR(20) UNIQUE,                          -- Número asignado al laboratorio (único)
  capacidad INT                                   -- Capacidad máxima del laboratorio
);

-- 4. Crear la tabla 'asignatura'
-- Esta tabla almacena las asignaturas que se imparten en los laboratorios
CREATE TABLE asignatura (
  id_asignatura INT AUTO_INCREMENT PRIMARY KEY,  -- Identificador único de la asignatura (clave primaria)
  nombre VARCHAR(100) UNIQUE,                           -- Nombre de la asignatura
  codigo VARCHAR(20) UNIQUE                      -- Código único de la asignatura
);

-- 5. Crear la tabla 'horario_laboratorio'
-- Esta tabla relaciona laboratorios, asignaturas, horarios y docentes
CREATE TABLE horario_laboratorio (
  id_horario INT AUTO_INCREMENT PRIMARY KEY,      -- Identificador único del horario (clave primaria)
  id_laboratorio INT,                             -- Clave foránea al laboratorio
  id_asignatura INT,                              -- Clave foránea a la asignatura
  dia VARCHAR(100),  -- Día de la semana
  horario_inicio VARCHAR(50),                            -- Hora de inicio de la clase
  horario_fin VARCHAR(50),                               -- Hora de fin de la clase
  id_usuario INT,                                 -- Clave foránea al docente (usuario)
  codigo_horario VARCHAR(300) UNIQUE,			  -- Almacenar el código del horario (unico)
  FOREIGN KEY (id_laboratorio) REFERENCES laboratorio(id_laboratorio) ON DELETE CASCADE,   -- Si se elimina el laboratorio, se eliminan los horarios asociados
  FOREIGN KEY (id_asignatura) REFERENCES asignatura(id_asignatura) ON DELETE SET NULL,     -- Si se elimina la asignatura, se establece NULL en este campo
  FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario) ON DELETE SET NULL               -- Si se elimina el docente, se establece NULL en este campo
);

-- 6. Crear la tabla 'equipo'
-- Esta tabla almacena los equipos disponibles en los laboratorios
CREATE TABLE equipo (
  id_equipo INT AUTO_INCREMENT PRIMARY KEY,     -- Identificador único del equipo (clave primaria)
  cod_patrimonial VARCHAR(50) UNIQUE,           -- Código patrimonial del equipo (único)
  id_laboratorio INT,                           -- Clave foránea al laboratorio donde está el equipo
  tipo_equipo ENUM('Teclado', 'CPU', 'Monitor', 'PizarraDigital'),                     -- Tipo o categoría del equipo (ejemplo: Computadora, Proyector)
  numero_serie VARCHAR(50),                     -- Número de serie del equipo
  estado VARCHAR(50),     -- Estado actual del equipo
  FOREIGN KEY (id_laboratorio) REFERENCES laboratorio(id_laboratorio) ON DELETE SET NULL  -- Si se elimina el laboratorio, el campo se establece en NULL
);

-- 7. Crear la tabla 'asistencia'
-- Esta tabla registra la asistencia de los alumnos a las clases en los laboratorios
CREATE TABLE asistencia (
  id_asistencia INT AUTO_INCREMENT PRIMARY KEY,  -- Identificador único de la asistencia (clave primaria)
  fecha DATE,                                    -- Fecha de la asistencia
  estado ENUM('PRESENTE', 'AUSENTE'),            -- Estado de asistencia del alumno
  id_alumno INT,                                 -- Clave foránea al alumno
  id_horario INT,                                -- Clave foránea al horario de la clase
  FOREIGN KEY (id_alumno) REFERENCES alumno(id_alumno) ON DELETE CASCADE,                -- Si se elimina el alumno, se eliminan sus asistencias
  FOREIGN KEY (id_horario) REFERENCES horario_laboratorio(id_horario) ON DELETE CASCADE, -- Si se elimina el horario, se eliminan las asistencias asociadas
  CONSTRAINT unique_alumno_horario_fecha UNIQUE (id_alumno, id_horario, fecha)  -- Combinación de alumno, horario y fecha debe ser única
);
